namespace MyApp.Core;

using Module = System.Console;

public enum Status { Pending, Completed }
public delegate void UpdateHandler(string message);

public interface IProcessable<T> where T : struct
{
    event UpdateHandler? OnUpdate;
    T Data { get; }
}

// Using a record struct for an immutable data type
public readonly record struct Point(int X, int Y)
{
    public static Point operator +(Point a, Point b) => new(a.X + b.X, a.Y + b.Y);
}

public class Calculator
{
    public void RunExample()
    {
        // Two instances of the Point struct
        var point1 = new Point(10, 20);
        var point2 = new Point(5, 8);

        // Using the overloaded '+' operator
        Point result = point1 + point2;

        Module.WriteLine($"Result: X={result.X}, Y={result.Y}");
    }
}

// Using primary constructors for dependency injection
public class DataService<T>(ILogger logger) where T : struct
{
    private readonly Guid _id = Guid.NewGuid();
    public required string Name { get; init; }

    public void Process(T? data)
    {
        // Using extension method
        logger.GetTypeName();

        const double PI = 3.14;
        var message = data is not null ? this.GetTypeName() : "No data";
        
        // Using a raw, interpolated string literal
        var log = $$"""
            Processing "{message}"
            ID: {_id}
            PI: {PI}
            """;
        
        // Using a verbatim string literal
        string? verbatim = @"C:\Path\To\File.txt";
        
        // Using string escape character
        Module.WriteLine("C:\\Path\\To\\File.txt");

        Module.WriteLine(log);
        Module.WriteLine(verbatim);
    }

[System.Diagnostics.CodeAnalysis.ExcludeFromCodeCoverage]
public static class ObjectExtensions
{
    public static string GetTypeName(this object _) => "System.Object";
}
